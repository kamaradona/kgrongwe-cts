Google Sheets Integration Setup ✓ CONNECTED

Once connected, camera traps, species settings, and all sighting entries sync automatically across every team member's laptop. One person registers traps — everyone sees them instantly. Sightings logged on any laptop appear on all others within 60 seconds. Setup takes ~5 minutes, done once.

1
Create a new Google Sheet
Go to sheets.google.com and create a new blank spreadsheet. Name it Karongwe Camera Trap Log. Leave it open in a tab.
2
Open Apps Script
In the Google Sheet, click Extensions → Apps Script. Delete any existing code in the editor, paste in the entire script below, then click Save (Ctrl+S).
// Karongwe Camera Trap System — Cloud Sync Backend // Handles traps, species whitelist, and sightings across all team laptops const SHEETS = { traps: "Traps", species: "Species", sightings: "Sightings" }; function doGet(e) { const action = e?.parameter?.action || "ping"; try { if (action === "ping") return json({ ok: true, message: "Karongwe CTS webhook active" }); if (action === "pull") return json({ ok: true, traps: readSheet(SHEETS.traps), species: readSheet(SHEETS.species), sightings: readSheet(SHEETS.sightings) }); return json({ ok: false, error: "Unknown action" }); } catch(err) { return json({ ok: false, error: err.message }); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action; if (action === "save_traps") { writeSheet(SHEETS.traps, ["id","name","location","lat","lng","habitat","notes"], body.traps); return json({ ok: true }); } if (action === "save_species") { writeSheet(SHEETS.species, ["name","category","enabled"], body.species); return json({ ok: true }); } if (action === "append_sightings") { appendRows(SHEETS.sightings, ["id","date","trap_id","habitat","lat","lng","species","species_scientific", "sex","age_class","count","time_of_day","behaviour","observer","ai_notes", "species_confidence","sex_confidence","age_confidence","logged_at","logged_by"], body.rows); return json({ ok: true }); } if (action === "delete_sighting") { deleteSighting(body.id); return json({ ok: true }); } return json({ ok: false, error: "Unknown action" }); } catch(err) { return json({ ok: false, error: err.message }); } } function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.getRange(1,1,1,headers.length).setValues([headers]); sheet.getRange(1,1,1,headers.length).setBackground("#3d2b1f").setFontColor("#c4a35a").setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } function readSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(name); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0]; return data.slice(1).map(row => { const obj={}; headers.forEach((h,i)=>obj[h]=row[i]); return obj; }); } function writeSheet(name, headers, rows) { const sheet = getOrCreateSheet(name, headers); if (sheet.getLastRow() > 1) sheet.getRange(2,1,sheet.getLastRow()-1,headers.length).clearContent(); if (rows && rows.length > 0) { const values = rows.map(r => headers.map(h => r[h]!==undefined ? r[h] : "")); sheet.getRange(2,1,values.length,headers.length).setValues(values); } } function appendRows(name, headers, rows) { const sheet = getOrCreateSheet(name, headers); if (!rows || !rows.length) return; const values = rows.map(r => headers.map(h => r[h]!==undefined ? r[h] : "")); sheet.getRange(sheet.getLastRow()+1,1,values.length,headers.length).setValues(values); } function deleteSighting(id) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEETS.sightings); if (!sheet) return; const data = sheet.getDataRange().getValues(); for (let i=data.length-1;i>=1;i--) { if (String(data[i][0])===String(id)) { sheet.deleteRow(i+1); break; } } } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
3
Deploy as a Web App
In Apps Script, click Deploy → New deployment.
Click the gear icon next to "Type" and select Web app.
Set Execute as: Me
Set Who has access: Anyone
Click Deploy — authorise the permissions when prompted (this is safe, it's your own script).

Copy the Web app URL that appears after deployment.
4
Paste your Web App URL — do this on every team laptop
Paste the URL Google gave you and click Save & Test:
5
Done — you're connected
Go to the Sighting Log tab and click ↑ Push to Sheets to sync. New rows are always appended — the sheet is never overwritten. You can push as often as you like; already-pushed entries are tracked and won't be duplicated.
